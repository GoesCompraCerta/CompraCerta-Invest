Yahoo Finance - Módulo de busca de cotações americanas

Este módulo foi implementado em Node no backend do projeto e consulta a API pública estável do Yahoo Finance.

Trecho completo:
const normalizeYahooSymbol = (value) => String(value || '')
  .trim()
  .toUpperCase()
  .replace(/\s+/g, '')
  .replace(/\./g, '-')
  .replace(/-+/g, '-');

const handleYahooRequest = async (request, response) => {
  const requestUrl = new URL(request.url, `http://${request.headers.host}`);
  const symbol = normalizeYahooSymbol(requestUrl.searchParams.get('symbol'));

  if (!symbol || !/^[A-Z0-9.^=-]+$/.test(symbol)) {
    sendJson(response, 400, { error: 'Ticker Yahoo Finance inválido.' });
    return;
  }

  try {
    const yahooUrl = new URL(`${YAHOO_BASE_URL}/${encodeURIComponent(symbol)}`);
    yahooUrl.searchParams.set('range', '1d');
    yahooUrl.searchParams.set('interval', '1d');
    yahooUrl.searchParams.set('events', 'div,splits');

    const upstream = await fetch(yahooUrl, {
      headers: { Accept: 'application/json', 'User-Agent': 'Mozilla/5.0' }
    });

    if (!upstream.ok) {
      const status = upstream.status === 404 ? 404 : 502;
      const message = upstream.status === 404
        ? `Ticker ${symbol} não encontrado no Yahoo Finance.`
        : `Yahoo Finance HTTP ${upstream.status}`;
      sendJson(response, status, { error: message, code: upstream.status === 404 ? 'TICKER_NOT_FOUND' : 'UPSTREAM_ERROR' });
      return;
    }

    const payload = await upstream.json();
    const result = payload?.chart?.result?.[0];
    const price = Number(result?.meta?.regularMarketPrice ?? result?.indicators?.quote?.[0]?.close?.at(-1));

    if (!Number.isFinite(price) || price <= 0) {
      sendJson(response, 502, { error: 'Yahoo Finance retornou uma cotação inválida.' });
      return;
    }

    sendJson(response, 200, { symbol, price, currency: result?.meta?.currency || 'USD' });
  } catch (error) {
    sendJson(response, 502, { error: error.message || 'Falha ao consultar o Yahoo Finance.' });
  }
};
